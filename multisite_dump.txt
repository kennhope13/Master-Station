function parseLocation(raw?: any): StationLocation {
  if (!raw) return {};
  if (typeof raw === 'object') return raw;
  try { return JSON.parse(raw); } catch { return {}; }
}

                      <div style={{ fontSize: 8, fontWeight: 800, color: 'var(--admin-text-muted)', marginBottom: 4, textTransform: 'uppercase' }}>
                        Alerts
                      </div>
                      {selectedView.alertsList.length === 0 ? (
                        <div style={{
                          flex: 1, display: 'flex', flexDirection: 'column', alignItems: 'center', justifyContent: 'center',
                          background: 'rgba(16,185,129,0.02)', border: '1px dashed rgba(16,185,129,0.2)', padding: 10
                        }}>
                          <ShieldCheck size={20} style={{ color: 'var(--admin-success)' }} />
                        </div>
                      ) : (
                        <div style={{ display: 'flex', flexDirection: 'column', gap: 4, overflowY: 'auto', maxHeight: 200 }} className="custom-hud-scroll">
                          {selectedView.alertsList.map(a => {
                            const isAlarm = a.level === 'alarm' || a.level === 'danger';
                            return (
                              <div
                                key={a.id}
                                style={{
                                  background: 'var(--admin-hover)',
                                  borderLeft: `2px solid ${isAlarm ? 'var(--admin-danger)' : 'var(--admin-warning)'}`,
                                  padding: '4px', display: 'flex', flexDirection: 'column'
                                }}
                              >
                                <div style={{ display: 'flex', justifyContent: 'space-between', fontSize: 7, color: 'var(--admin-text-muted)' }}>
                                  <span style={{ fontWeight: 800, color: isAlarm ? 'var(--admin-danger)' : 'var(--admin-warning)' }}>
                                    {isAlarm ? 'ALARM' : 'WARN'}
                                  </span>
                                </div>
                                <div style={{ fontSize: '0.6rem', color: 'var(--admin-text)', fontWeight: 600, overflow: 'hidden', textOverflow: 'ellipsis' }}>
                                  {a.message}
                                </div>
                              </div>
                            );
                          })}
                        </div>
                      )}
                    </div>
                  </div>
                </div>
              </div>
            </div>
          )}

          {views.length === 0 && (
            <div style={{
              position: 'absolute', top: '50%', left: '50%', transform: 'translate(-50%, -50%)',
              zIndex: 1000, color: 'var(--admin-text-muted)', textAlign: 'center',
              background: 'var(--admin-overlay)', padding: '24px 36px', border: '1px solid var(--admin-border)',
              boxShadow: '0 10px 40px rgba(0,0,0,0.5)', borderRadius: 4, backdropFilter: 'blur(10px)'
            }}>
              <AlertTriangle size={36} style={{ color: 'var(--admin-warning)', marginBottom: 12, display: 'inline-block' }} />
              <h3 style={{ color: 'var(--admin-text)', margin: '0 0 6px 0', fontSize: '0.85rem' }}>Chưa Cập Nhật Trạm Biến Áp</h3>
              <p style={{ margin: 0, fontSize: '0.75rem' }}>Vui lòng khởi tạo trạm trong giao diện Quản lý hệ thống.</p>
            </div>
          )}
        </>
      )}

      {/* ADD STATION MODAL OVERLAY */}
      {isAddModalOpen && (
        <div style={{
          position: 'fixed', inset: 0, zIndex: 9999,
          background: 'rgba(0, 0, 0, 0.65)', backdropFilter: 'blur(4px)',
          display: 'flex', alignItems: 'center', justifyContent: 'center'
        }}>
          <div style={{
            background: 'var(--admin-panel)',
            border: '1px solid var(--admin-border)',
            width: '90%', maxWidth: '440px',
            boxShadow: '0 20px 50px rgba(0, 0, 0, 0.4)',
            color: 'var(--admin-text)'
          }}>
            {/* Modal Header */}
            <div style={{
              background: 'var(--admin-bg)',
              padding: '14px 20px',
              borderBottom: '1px solid var(--admin-border)',
              display: 'flex', alignItems: 'center', justifyContent: 'space-between'
            }}>
              <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
                <Plus size={16} style={{ color: 'var(--admin-accent)' }} />
                <span style={{ fontSize: '0.8rem', fontWeight: 800, textTransform: 'uppercase', letterSpacing: 0.5 }}>
                  Thêm trạm biến áp mới
                </span>
              </div>
              <button 
                onClick={() => setIsAddModalOpen(false)}
                style={{ background: 'none', border: 'none', color: 'var(--admin-text-muted)', cursor: 'pointer' }}
              >
                <X size={16} />
              </button>
            </div>

            {/* Modal Body */}
            <div style={{ padding: '20px', display: 'flex', flexDirection: 'column', gap: 14 }}>
              <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                <label style={{ fontSize: '0.68rem', fontWeight: 700, color: 'var(--admin-text-muted)' }}>TÊN TRẠM BIẾN ÁP *</label>
                <input 
                  type="text" 
                  placeholder="Ví dụ: Trạm 110kV Cần Thơ"
                  value={newStationName}
                  onChange={e => setNewStationName(e.target.value)}
                  style={{
                    background: 'var(--admin-layer-2)', border: '1px solid var(--admin-border)',
                    padding: '8px 10px', fontSize: '0.75rem', color: 'var(--admin-text)', outline: 'none'
                  }}
                />
              </div>

              <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                <label style={{ fontSize: '0.68rem', fontWeight: 700, color: 'var(--admin-text-muted)' }}>MÃ TRẠM *</label>
                <input 
                  type="text" 
                  placeholder="Ví dụ: TBA-CT01"
                  value={newStationCode}
                  onChange={e => setNewStationCode(e.target.value)}
                  style={{
                    background: 'var(--admin-layer-2)', border: '1px solid var(--admin-border)',
                    padding: '8px 10px', fontSize: '0.75rem', color: 'var(--admin-text)', outline: 'none'
                  }}
                />
              </div>

              <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 10 }}>
                <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                  <label style={{ fontSize: '0.68rem', fontWeight: 700, color: 'var(--admin-text-muted)' }}>VĨ ĐỘ (LAT) *</label>
                  <input 
                    type="number" 
                    step="0.000001"
                    placeholder="Ví dụ: 10.03"
                    value={newStationLat}
                    onChange={e => setNewStationLat(e.target.value)}
                    style={{
                      background: 'var(--admin-layer-2)', border: '1px solid var(--admin-border)',
                      padding: '8px 10px', fontSize: '0.75rem', color: 'var(--admin-text)', outline: 'none'
                    }}
                  />
                </div>
                <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                  <label style={{ fontSize: '0.68rem', fontWeight: 700, color: 'var(--admin-text-muted)' }}>KINH ĐỘ (LNG) *</label>
                  <input 
                    type="number" 
                    step="0.000001"
                    placeholder="Ví dụ: 105.78"
                    value={newStationLng}
                    onChange={e => setNewStationLng(e.target.value)}
                    style={{
                      background: 'var(--admin-layer-2)', border: '1px solid var(--admin-border)',
                      padding: '8px 10px', fontSize: '0.75rem', color: 'var(--admin-text)', outline: 'none'
                    }}
                  />
                </div>
              </div>

              <div style={{ display: 'flex', flexDirection: 'column', gap: 4 }}>
                <label style={{ fontSize: '0.68rem', fontWeight: 700, color: 'var(--admin-text-muted)' }}>ĐỊA CHỈ</label>
                <input 
                  type="text" 
                  placeholder="Ví dụ: Ninh Kiều, Cần Thơ"
                  value={newStationAddress}
                  onChange={e => setNewStationAddress(e.target.value)}
                  style={{
                    background: 'var(--admin-layer-2)', border: '1px solid var(--admin-border)',

              display: 'flex', justifyContent: 'flex-end', gap: 10
            }}>
              <button 
                onClick={() => setIsAddModalOpen(false)}
                disabled={isSaving}
                className="btn-industrial"
                style={{ padding: '6px 16px', fontSize: '0.75rem' }}
              >
                Hủy
              </button>
              <button 
                onClick={handleAddStationSubmit}
                disabled={isSaving}
                className="btn-industrial btn-primary"
                style={{
                  padding: '6px 16px', fontSize: '0.75rem', fontWeight: 700,
                  background: 'var(--admin-accent)', color: '#fff', border: 'none'

          <div className="modal-content" style={{ width: 360, textAlign: 'center' }} onClick={e => e.stopPropagation()}>
            <div className="modal-body" style={{ padding: '32px 24px' }}>

              </div>
            </div>
          </div>
        </div>
      )}
    </div>
  );
}

function CentralReportsView({
  stations, views, globalStats, selectedStationId,
}: {
  stations: Station[];
  views: StationView[];
  globalStats: { totalStations: number; totalDevices: number; onlineDevices: number; totalAlerts: number; totalCameras: number };
  selectedStationId: string | null;
}) {
  const [scopeStationId, setScopeStationId] = useState(selectedStationId || '');
  const [history, setHistory] = useState<ReportItem[]>([]);
  const [historyLoading, setHistoryLoading] = useState(false);
  const [generatingType, setGeneratingType] = useState<string | null>(null);
  const [stationSearch, setStationSearch] = useState('');
  const [historyFilter, setHistoryFilter] = useState('all');
  const [stationMenuOpen, setStationMenuOpen] = useState(false);
  const [menuPos, setMenuPos] = useState({ top: 0, left: 0, width: 0 });
  const stationBtnRef = useRef<HTMLButtonElement>(null);

  useEffect(() => { if (selectedStationId) setScopeStationId(selectedStationId); }, [selectedStationId]);

  const scopeView = useMemo(() => views.find(v => v.station.id === scopeStationId) || null, [views, scopeStationId]);
  const scopeLabel = scopeView?.station.name || 'Toàn bộ hệ thống';

  const summary = useMemo(() => {
    if (scopeView) return { stationCount: 1, devices: scopeView.kpi.devicesTotal, online: scopeView.kpi.devicesOnline, alerts: scopeView.kpi.alerts };
    return { stationCount: globalStats.totalStations, devices: globalStats.totalDevices, online: globalStats.onlineDevices, alerts: globalStats.totalAlerts };
  }, [globalStats, scopeView]);

  const loadHistory = async () => {
    setHistoryLoading(true);
    try {
      const items = await stationApi.getReports(scopeStationId || undefined);
      setHistory(items.sort((a, b) => new Date(b.generatedAt).getTime() - new Date(a.generatedAt).getTime()));
    } catch { setHistory([]); } finally { setHistoryLoading(false); }
  };
  useEffect(() => { loadHistory(); }, [scopeStationId]);

  const generateQuickReport = async (type: 'daily' | 'monthly' | 'event') => {
    const now = new Date(), from = new Date(now);
    if (type === 'daily') from.setDate(now.getDate() - 1);
    if (type === 'monthly') from.setDate(now.getDate() - 30);
    if (type === 'event') from.setDate(now.getDate() - 7);
    setGeneratingType(type);
    try {
      await stationApi.generateReport({ stationId: scopeStationId || undefined, type, from: from.toISOString(), to: now.toISOString() });
      await loadHistory();
    } catch { alert('Không thể tạo báo cáo.'); } finally { setGeneratingType(null); }
  };

  const downloadReportFile = async (report: ReportItem) => {
    try {
      const blob = await stationApi.downloadReport(report.id);
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a'); a.href = url;
      a.download = `BaoCao_${report.type}_${report.generatedAt.slice(0, 10)}.pdf`; a.click();
      URL.revokeObjectURL(url);
    } catch { alert('Không thể tải báo cáo.'); }
  };

  const deleteReportItem = async (id: string) => { try { await stationApi.deleteReport(id); loadHistory(); } catch {} };

  const exportAlertsCsv = async () => {
    try {
      const blob = await stationApi.exportAlertsCsv({ status: 'open' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a'); a.href = url;
      a.download = `CanhBao_DaTram_${new Date().toISOString().slice(0, 10)}.csv`; a.click();
      URL.revokeObjectURL(url);
    } catch { alert('Không thể xuất CSV.'); }
  };

  const filteredViews = useMemo(() => {
    if (!stationSearch) return views;
    return views.filter(v => v.station.name.toLowerCase().includes(stationSearch.toLowerCase()));
  }, [views, stationSearch]);

  const typeLabel = (t: string) => t === 'daily' ? 'Hàng ngày' : t === 'monthly' ? 'Hàng tháng' : 'Sự cố';
  const typeColor = (t: string) => t === 'daily' ? 'var(--admin-accent)' : t === 'monthly' ? '#a855f7' : '#f97316';
  const filteredHistory = useMemo(() => historyFilter === 'all' ? history : history.filter(r => r.type === historyFilter), [history, historyFilter]);

  const reportCards = [
    { type: 'daily' as const, title: 'Báo cáo ngày', icon: <Clock size={12} />, color: 'var(--admin-accent)' },
    { type: 'monthly' as const, title: 'Báo cáo tháng', icon: <Calendar size={12} />, color: '#a855f7' },
    { type: 'event' as const, title: 'Báo cáo sự cố', icon: <AlertTriangle size={12} />, color: '#f97316' },
  ];

  return (
    <div className="crv-page" style={{ padding: '10px 16px', gap: 10 }}>

      {/* Quick generate */}
      <div className="crv-gen" style={{ padding: '6px 12px', borderRadius: 3 }}>
        <span className="crv-gen-lbl">Tạo nhanh:</span>
        <div className="crv-gen-sep" />
        {reportCards.map(card => (
          <button key={card.type} className={`crv-gen-btn ${generatingType === card.type ? 'generating' : ''}`}
            style={{ '--crv-color': card.color } as React.CSSProperties}
            disabled={!!generatingType}
            onClick={() => !generatingType && generateQuickReport(card.type)}>
            <span className="crv-gen-dot" />
            {generatingType === card.type
              ? <><Loader2 size={12} className="crv-spin" /> Đang tạo...</>
              : <>{card.icon && <span style={{ color: card.color, display: 'flex' }}>{card.icon}</span>} {card.title}</>}
          </button>
        ))}
        <div style={{ flex: 1 }} />
        <button className="btn-industrial btn-sm" style={{ display: 'inline-flex', alignItems: 'center', gap: 5, height: 28 }} onClick={loadHistory}>
          <RefreshCw size={11} />
        </button>
      </div>

      {/* Scope + filters */}
      <div style={{ display: 'flex', alignItems: 'center', gap: 10 }}>
        <div style={{ display: 'flex', alignItems: 'center', gap: 6 }}>
          <span style={{ fontSize: '.58rem', fontWeight: 900, color: 'var(--admin-text-muted)', letterSpacing: '.1em', textTransform: 'uppercase' }}>Phạm vi</span>
          <div style={{ position: 'relative' }}>
            {stationMenuOpen && <div style={{ position: 'fixed', inset: 0, zIndex: 9998 }} onClick={() => setStationMenuOpen(false)} />}
            <button ref={stationBtnRef} className="crv-station-btn" style={{ width: 200, height: 30, fontSize: '.65rem' }}
              onClick={() => {
                const r = stationBtnRef.current?.getBoundingClientRect();
                if (r) setMenuPos({ top: r.bottom, left: r.left, width: r.width });
                setStationMenuOpen(o => !o);
              }}>
              <span style={{ overflow: 'hidden', textOverflow: 'ellipsis', whiteSpace: 'nowrap' }}>{scopeLabel}</span>
              <span style={{ flexShrink: 0, opacity: 0.5, fontSize: '.6rem' }}>▾</span>
            </button>
            {stationMenuOpen && (
              <div className="crv-station-menu" style={{ top: menuPos.top, left: menuPos.left, width: menuPos.width }}>
                <div className={`crv-station-opt ${!scopeStationId ? 'active' : ''}`} onClick={() => { setScopeStationId(''); setStationMenuOpen(false); }}>
                  Toàn bộ hệ thống ({views.length} trạm)
                </div>
                {views.map(v => (
                  <div key={v.station.id} className={`crv-station-opt ${scopeStationId === v.station.id ? 'active' : ''}`}
                    onClick={() => { setScopeStationId(v.station.id); setStationMenuOpen(false); }}>
                    {(v.station.code ? `${v.station.code} · ` : '') + v.station.name}
                  </div>
                ))}
              </div>
            )}
          </div>
        </div>
        <div style={{ flex: 1 }} />
        <span style={{ fontSize: '.58rem', color: 'var(--admin-text-muted)', fontWeight: 700 }}>{filteredHistory.length} báo cáo</span>
        {(['all', 'daily', 'monthly', 'event'] as const).map(f => (
          <button key={f} className={`crv-filter-btn ${historyFilter === f ? 'active' : ''}`} onClick={() => setHistoryFilter(f)}>
            {f === 'all' ? 'Tất cả' : typeLabel(f)}
          </button>
        ))}
      </div>

      {/* Table */}
      <div style={{ flex: 1, overflow: 'hidden', display: 'flex', flexDirection: 'column', minHeight: 0 }}>
        <div className="crv-table-wrap">
          {historyLoading ? (
            <div className="crv-empty"><Loader2 size={26} className="crv-spin" style={{ color: 'var(--admin-accent)', opacity: 1 }} /><p>Đang tải...</p></div>
          ) : filteredHistory.length === 0 ? (
            <div className="crv-empty"><FileText size={38} /><p>{history.length === 0 ? 'Chưa có báo cáo nào.' : 'Không có báo cáo phù hợp.'}</p></div>
          ) : (
            <table className="crv-table">
              <thead><tr>
                <th style={{ width: 100 }}>Loại</th><th>Trạm</th><th style={{ width: 170 }}>Kỳ báo cáo</th><th style={{ width: 130 }}>Ngày tạo</th><th style={{ width: 90, textAlign: 'right' }}></th>
              </tr></thead>
              <tbody>
                {filteredHistory.map(item => (
                  <tr key={item.id}>
                    <td><div className="crv-type-pill"><span className="crv-type-bar" style={{ background: typeColor(item.type) }} /><span className="crv-type-lbl">{typeLabel(item.type)}</span></div></td>
                    <td>{stations.find(s => s.id === item.stationId)?.name || <span style={{ color: 'var(--admin-text-muted)', fontStyle: 'italic' }}>Toàn hệ thống</span>}</td>
                    <td style={{ color: 'var(--admin-text-muted)', fontFamily: 'monospace', fontSize: '.68rem' }}>
                      {item.periodFrom ? new Date(item.periodFrom).toLocaleDateString('vi-VN') : '--'}
                      {' → '}{item.periodTo ? new Date(item.periodTo).toLocaleDateString('vi-VN') : '--'}
                    </td>
                    <td style={{ color: 'var(--admin-text-muted)', fontSize: '.72rem' }}>{new Date(item.generatedAt).toLocaleString('vi-VN')}</td>
                    <td style={{ textAlign: 'right', whiteSpace: 'nowrap' }}>
                      {item.fileUrl && (
                        <button className="btn-industrial btn-sm" style={{ display: 'inline-flex', alignItems: 'center', gap: 5, marginRight: 6 }} onClick={() => downloadReportFile(item)}>
                          <Download size={11} /> PDF
                        </button>
                      )}
                      <button className="crv-del-btn" onClick={() => deleteReportItem(item.id)} title="Xóa"><X size={12} /></button>
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          )}
        </div>
      </div>
    </div>
  );
}
          {stationMenuOpen && (
            <div className="crv-station-menu" style={{ top: menuPos.top, left: menuPos.left, width: menuPos.width }}>
              <div className={`crv-station-opt ${!scopeStationId ? 'active' : ''}`} onClick={() => { setScopeStationId(''); setStationMenuOpen(false); }}>
                Toàn bộ hệ thống ({views.length} trạm)
              </div>
              {views.map(v => (
                <div key={v.station.id} className={`crv-station-opt ${scopeStationId === v.station.id ? 'active' : ''}`}
                  onClick={() => { setScopeStationId(v.station.id); setStationMenuOpen(false); }}>
                  {(v.station.code ? `${v.station.code} · ` : '') + v.station.name}
                </div>
              ))}
            </div>
          )}
        </div>

        <div style={{ display: 'flex', alignItems: 'center', gap: 14, paddingLeft: 10, borderLeft: '1px solid var(--admin-border)' }}>
          <span className="crv-stat-chip">TRẠM <b>{summary.stationCount}</b></span>
          <span className="crv-stat-chip">ONLINE <b style={{ color: summary.online === summary.devices ? 'var(--admin-success)' : '#f59e0b' }}>{summary.online}/{summary.devices}</b></span>
          <span className="crv-stat-chip">CẢNH BÁO <b style={{ color: summary.alerts > 0 ? 'var(--admin-danger)' : 'var(--admin-success)' }}>{summary.alerts}</b></span>
          <span className="crv-stat-chip">BÁO CÁO <b>{history.length}</b></span>
        </div>

        <div style={{ flex: 1 }} />
        <div style={{ display: 'flex', alignItems: 'center', gap: 8 }}>
          <button className="btn-industrial btn-sm" style={{ display: 'inline-flex', alignItems: 'center', gap: 5, height: 28 }} onClick={loadHistory}>
            <RefreshCw size={11} /> Làm mới
          </button>
          <button className="btn-industrial btn-primary btn-sm" style={{ display: 'inline-flex', alignItems: 'center', gap: 5, height: 28 }} onClick={exportAlertsCsv}>
            <Download size={11} /> Xuất CSV
          </button>
        </div>
      </div>

      <div className="crv-gen">
        <span className="crv-gen-lbl">Tạo báo cáo nhanh:</span>
        <div className="crv-gen-sep" />
        {reportCards.map(card => (
          <button key={card.type} className={`crv-gen-btn ${generatingType === card.type ? 'generating' : ''}`}
            style={{ '--crv-color': card.color } as React.CSSProperties}
            disabled={!!generatingType}
            onClick={() => !generatingType && generateQuickReport(card.type)}>
            <span className="crv-gen-dot" />
            {generatingType === card.type
              ? <><Loader2 size={12} className="crv-spin" /> Đang tạo...</>
              : <>{card.icon && <span style={{ color: card.color, display: 'flex' }}>{card.icon}</span>} {card.title}</>}
          </button>
        ))}
      </div>

      <div className="crv-main">
        <div className="crv-slist">
          <div className="crv-slist-hd"><Map size={11} /><span>Phạm vi</span></div>
          <div className="crv-slist-search">
            <Search size={11} />
            <input type="text" placeholder="Tìm trạm..." value={stationSearch} onChange={e => setStationSearch(e.target.value)} />
          </div>
          <div className="crv-slist-body">
            <div className={`crv-sitem ${!scopeStationId ? 'active' : ''}`} onClick={() => setScopeStationId('')}>
              <div className="crv-sitem-dot" style={{ background: 'var(--admin-accent)' }} />
              <div className="crv-sitem-info"><span className="n">Toàn bộ hệ thống</span><span className="s">{views.length} trạm</span></div>
            </div>
            {filteredViews.map(view => {
              const active = scopeStationId === view.station.id;
              return (
                <div key={view.station.id} className={`crv-sitem ${active ? 'active' : ''}`} onClick={() => setScopeStationId(view.station.id)}>
                  <div className="crv-sitem-dot" style={{ background: view.kpi.devicesOnline > 0 ? '#10b981' : '#6b7280', boxShadow: view.kpi.devicesOnline > 0 ? '0 0 4px #10b981' : 'none' }} />
                  <div className="crv-sitem-info">
                    <span className="n">{view.station.name}</span>
                    <span className="s">{view.station.code || view.station.id.slice(0, 8).toUpperCase()}{' · '}{view.kpi.devicesOnline}/{view.kpi.devicesTotal}</span>
                  </div>
                  {view.kpi.alerts > 0 && <span className="crv-sitem-badge">{view.kpi.alerts}</span>}
                </div>
              );
            })}
          </div>
        </div>

        <div className="crv-content">
          <div className="crv-content-hd">
            <span className="crv-content-hd-title">Lịch sử báo cáo</span>
            <span className="crv-count-badge">{filteredHistory.length}</span>
            <span className="crv-scope-label">· {scopeStationId ? scopeLabel : 'Toàn hệ thống'}</span>
            <div style={{ flex: 1 }} />
            {(['all', 'daily', 'monthly', 'event'] as const).map(f => (
              <button key={f} className={`crv-filter-btn ${historyFilter === f ? 'active' : ''}`} onClick={() => setHistoryFilter(f)}>
                {f === 'all' ? 'Tất cả' : typeLabel(f)}
              </button>
            ))}
          </div>

          <div className="crv-table-wrap">
            {historyLoading ? (
              <div className="crv-empty"><Loader2 size={26} className="crv-spin" style={{ color: 'var(--admin-accent)', opacity: 1 }} /><p>Đang tải...</p></div>
            ) : filteredHistory.length === 0 ? (
              <div className="crv-empty"><FileText size={38} /><p>{history.length === 0 ? 'Chưa có báo cáo nào.' : 'Không có báo cáo phù hợp.'}</p></div>
            ) : (
              <table className="crv-table">
                <thead><tr>
                  <th>Loại</th><th>Trạm / Phạm vi</th><th>Kỳ báo cáo</th><th>Ngày tạo</th><th style={{ textAlign: 'right' }}></th>
                </tr></thead>
                <tbody>
                  {filteredHistory.map(item => (
                    <tr key={item.id}>
                      <td><div className="crv-type-pill"><span className="crv-type-bar" style={{ background: typeColor(item.type) }} /><span className="crv-type-lbl">{typeLabel(item.type)}</span></div></td>
                      <td>{stations.find(s => s.id === item.stationId)?.name || <span style={{ color: 'var(--admin-text-muted)', fontStyle: 'italic' }}>Toàn hệ thống</span>}</td>
                      <td style={{ color: 'var(--admin-text-muted)', fontFamily: 'monospace', fontSize: '.7rem' }}>
                        {item.periodFrom ? new Date(item.periodFrom).toLocaleDateString('vi-VN') : '--'}
                        {' → '}{item.periodTo ? new Date(item.periodTo).toLocaleDateString('vi-VN') : '--'}
                      </td>
                      <td style={{ color: 'var(--admin-text-muted)', fontSize: '.75rem' }}>{new Date(item.generatedAt).toLocaleString('vi-VN')}</td>
                      <td style={{ textAlign: 'right', whiteSpace: 'nowrap' }}>
                        {item.fileUrl && (
                          <button className="btn-industrial btn-sm" style={{ display: 'inline-flex', alignItems: 'center', gap: 5, marginRight: 6 }} onClick={() => downloadReportFile(item)}>
                            <Download size={11} /> PDF
                          </button>
                        )}
                        <button className="crv-del-btn" onClick={() => deleteReportItem(item.id)} title="Xóa"><X size={12} /></button>
                      </td>
                    </tr>
                  ))}
                </tbody>
              </table>
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
